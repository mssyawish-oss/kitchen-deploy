// --- per-ROW spinning detection on the cam feed. Two key guards:
//   (1) FROZEN-FEED: a live camera always has tiny sensor noise frame-to-frame, so the global diff is never
//       exactly ~0. If it IS ~0 the feed is FROZEN/LAGGED (not a real stop) → never alarm. This kills the
//       false "not spinning" that fired whenever the feed hiccuped.
//   (2) PER-ROW: each spit band is checked on its own, so 1 row can be spinning and another flagged stopped.
const _ROT_SPIN_BANDS=[[0.005,0.155],[0.165,0.300],[0.328,0.453],[0.465,0.580],[0.590,0.695],[0.705,0.795]];
const _ROT_SPIN_X=[0.21,0.81];
let _camSpinCanvas=null,_camSpinPrev=null,_camSpinStart=0,_camFeedLiveAt=0,_rowMotionAt=[0,0,0,0,0,0];
function _camSpinCheck(){
  const banner=document.getElementById('cam-stopped');
  if(appData.rotcam_public&&appData.rotcam_public.spin_enabled===false){   // alarm turned off in settings
    if(banner)banner.classList.remove('show');
    if(window._rotStopAlarm){window._rotStopAlarm=false;if(typeof stopAlarm==='function')stopAlarm('rot-stopped');}
    return;}
  const img=document.getElementById('rotcam-stream');
  if(!img||!img.complete||!img.naturalWidth)return;
  const W=128,H=72;
  const c=_camSpinCanvas||(_camSpinCanvas=document.createElement('canvas'));c.width=W;c.height=H;
  const ctx=c.getContext('2d',{willReadFrequently:true});
  let cur;try{ctx.drawImage(img,0,0,W,H);cur=ctx.getImageData(0,0,W,H).data;}catch(e){return;}
  const now=Date.now();
  if(!_camSpinPrev){_camSpinPrev=cur;_camSpinStart=now;_camFeedLiveAt=now;for(let r=0;r<6;r++)_rowMotionAt[r]=now;return;}
  let gd=0;for(let i=0;i<cur.length;i+=4)gd+=Math.abs(cur[i]-_camSpinPrev[i]);gd/=(cur.length/4);
  const FROZEN=0.25;                                      // <this = same frame repeated = feed frozen/lagging
  if(gd>=FROZEN)_camFeedLiveAt=now;                       // feed is genuinely live (has noise/motion)
  const x1=Math.floor(_ROT_SPIN_X[0]*W),x2=Math.ceil(_ROT_SPIN_X[1]*W);
  for(let r=0;r<6;r++){
    const ya=Math.floor(_ROT_SPIN_BANDS[r][0]*H),yb=Math.ceil(_ROT_SPIN_BANDS[r][1]*H);
    let d=0,n=0;for(let y=ya;y<yb;y++){for(let x=x1;x<x2;x++){const i=(y*W+x)*4;d+=Math.abs(cur[i]-_camSpinPrev[i]);n++;}}
    if(n&&(d/n)>2.2)_rowMotionAt[r]=now;                  // this spit band moved
  }
  _camSpinPrev=cur;
  const grace=(now-_camSpinStart)<8000;                  // settle after first frame
  const feedFresh=(now-_camFeedLiveAt)<5000;             // saw live noise/motion recently (NOT frozen)
  const pat=window._rotCPat||'';   // CONFIRMED occupancy (stable over 2+ reads) — a one-off glare misread must NOT trigger the alarm
  const loaded=r=>(pat.length===6&&pat[r]==='1');
  const stoppedRows=[];
  window._rowState=window._rowState||[];
  for(let r=0;r<6;r++){                                   // per-row state drives the animated panel too
    if(!loaded(r)) window._rowState[r]='empty';
    else if(!feedFresh||grace) window._rowState[r]='loaded';        // feed frozen/settling → can't judge spin
    else if((now-_rowMotionAt[r])>9000){ window._rowState[r]='stopped'; stoppedRows.push(r+1); }
    else window._rowState[r]='spinning';
  }
  const any=stoppedRows.length>0;
  if(banner){banner.classList.toggle('show',any);if(any)banner.textContent=(stoppedRows.length>1?'ROWS ':'ROW ')+stoppedRows.join(', ')+' NOT SPINNING';}
  if(any&&!window._rotStopAlarm&&!window._rotStopAcked){window._rotStopAlarm=true;if(typeof playAlarm==='function')playAlarm('rot-stopped');}
  else if(!any){window._rotStopAcked=false;if(window._rotStopAlarm){window._rotStopAlarm=false;if(typeof stopAlarm==='function')stopAlarm('rot-stopped');}}
}
function _doneForRow(row){
  const p=probeData[row],st=((probeStates[row]||{}).status)||'idle';
  const cooked=(typeof serverSettings!=='undefined'&&serverSettings&&serverSettings.cooked_temp)||75;
  const t=p?p.temp:null;
  if(t==null||(st==='idle'&&t<35))return{pct:0,color:'rgba(255,255,255,.22)',lab:'no probe'};
  const pct=Math.max(4,Math.min(100,Math.round((t/cooked)*100)));
  if(st==='overdone')return{pct,color:'#ff5b5b',lab:'OVERDONE'};
  if(st==='ready')return{pct,color:'#3ee0a6',lab:'READY'};
  if(st==='almost')return{pct,color:'#fbbf24',lab:'almost'};
  return{pct,color:'#5b9bff',lab:Math.round(t)+'°C'};
}
